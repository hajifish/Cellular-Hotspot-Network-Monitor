```swift
// RateHistory.swift（完整实现）
import Foundation

struct RateSample: Equatable {
    var rx: UInt64
    var tx: UInt64
}

/// 滚动速率历史：每秒一条增量，保留最近 capacity 条
final class RateHistory {
    let capacity: Int
    private(set) var samples: [RateSample] = []

    init(capacity: Int = 120) {
        precondition(capacity > 0)
        self.capacity = capacity
    }

    func push(_ sample: RateSample) {
        samples.append(sample)
        if samples.count > capacity {
            samples.removeFirst(samples.count - capacity)
        }
    }

    func clear() {
        samples.removeAll()
    }

    /// 最近 window 秒的平均速率（B/s）；样本数不足 window 时按实际条数平均
    func averageRX(window: Int) -> UInt64 { average({ $0.rx }, window: window) }
    func averageTX(window: Int) -> UInt64 { average({ $0.tx }, window: window) }

    private func average(_ field: (RateSample) -> UInt64, window: Int) -> UInt64 {
        let recent = samples.suffix(max(1, window))
        guard !recent.isEmpty else { return 0 }
        return recent.reduce(UInt64(0)) { $0 + field($1) } / UInt64(recent.count)
    }
}
```

```swift
// RateHistoryTests.swift（完整测试）
@testable import TrafficMonitor

enum RateHistoryTests {
    static func all() {
        pushKeepsCapacity()
        averageOverWindow()
        averageWithFewerSamples()
        clearEmpties()
        monitorFeedsHistory()
    }

    static func pushKeepsCapacity() {
        let history = RateHistory(capacity: 3)
        for i in 1...5 { history.push(RateSample(rx: UInt64(i), tx: 0)) }
        TestRun.expectEqual(history.samples.count, 3, "capacity kept")
        TestRun.expectEqual(history.samples.map(\.rx), [UInt64(3), 4, 5], "oldest dropped")
    }

    static func averageOverWindow() {
        let history = RateHistory(capacity: 10)
        for v in [UInt64(1_000), 2_000, 3_000, 4_000] { history.push(RateSample(rx: v, tx: 0)) }
        TestRun.expectEqual(history.averageRX(window: 2), UInt64(3_500), "avg last 2")
        TestRun.expectEqual(history.averageRX(window: 10), UInt64(2_500), "avg all 4")
    }

    static func averageWithFewerSamples() {
        let history = RateHistory()
        history.push(RateSample(rx: 900, tx: 0))
        TestRun.expectEqual(history.averageRX(window: 60), UInt64(900), "single-sample avg rx")
        TestRun.expectEqual(history.averageTX(window: 60), UInt64(0), "empty tx zero")
    }

    static func clearEmpties() {
        let history = RateHistory()
        history.push(RateSample(rx: 1, tx: 1))
        history.clear()
        TestRun.expect(history.samples.isEmpty, "cleared")
    }

    static func monitorFeedsHistory() {
        let monitor = TrafficMonitor(sampler: FakeSampler(samples: []))
        monitor.update(from: [ClassifiedSample(name: "en0", category: .wifi, counters: InterfaceCounters(rx: 0, tx: 0))])
        monitor.update(from: [ClassifiedSample(name: "en0", category: .wifi, counters: InterfaceCounters(rx: 500, tx: 250))])
        TestRun.expectEqual(monitor.history.samples.count, 1, "one delta per update")
        TestRun.expectEqual(monitor.history.samples.last, RateSample(rx: 500, tx: 250), "total delta recorded")
    }
}
```

```swift
import AppKit

final class RateGraphView: NSView {
    var history: RateHistory? { didSet { needsDisplay = true } }
    var downloadLabel = "" { didSet { needsDisplay = true } }
    var uploadLabel = "" { didSet { needsDisplay = true } }

    private static let downloadColor = NSColor(name: nil) { appearance in
        appearance.bestMatch(from: [.darkAqua, .aqua]) == .darkAqua
            ? NSColor(srgbRed: 0x39/255.0, green: 0x87/255.0, blue: 0xe5/255.0, alpha: 1) // #3987e5
            : NSColor(srgbRed: 0x2a/255.0, green: 0x78/255.0, blue: 0xd6/255.0, alpha: 1) // #2a78d6
    }
    private static let uploadColor = NSColor(name: nil) { appearance in
        appearance.bestMatch(from: [.darkAqua, .aqua]) == .darkAqua
            ? NSColor(srgbRed: 0xd9/255.0, green: 0x59/255.0, blue: 0x26/255.0, alpha: 1) // #d95926
            : NSColor(srgbRed: 0xeb/255.0, green: 0x68/255.0, blue: 0x34/255.0, alpha: 1) // #eb6834
    }

    /// 量程向上取整到 1/2/5×10^n（纯函数，可测）
    static func niceMaximum(_ value: UInt64) -> UInt64 {
        guard value > 0 else { return 1 }
        var magnitude = 1
        while UInt64(magnitude) * 10 <= value { magnitude *= 10 }
        for multiplier in [1, 2, 5, 10] {
            let candidate = UInt64(multiplier) * UInt64(magnitude)
            if candidate >= value { return candidate }
        }
        return value
    }

    override func draw(_ dirtyRect: NSRect) {
        super.draw(dirtyRect)
        let plot = bounds.insetBy(dx: 8, dy: 10)
        guard let history else { return }
        let samples = history.samples
        let maxRate = Self.niceMaximum(samples.map { max($0.rx, $0.tx) }.max() ?? 1)

        // 退隐网格：3 条水平细线 + 左上量程标注（文本色）
        NSColor.separatorColor.withAlphaComponent(0.35).setStroke()
        let grid = NSBezierPath()
        for fraction in [0.0, 0.5, 1.0] {
            let y = plot.minY + plot.height * CGFloat(fraction)
            grid.move(to: NSPoint(x: plot.minX, y: y))
            grid.line(to: NSPoint(x: plot.maxX, y: y))
        }
        grid.lineWidth = 0.5
        grid.stroke()
        drawCaption(ByteFormat.rate(maxRate), at: NSPoint(x: plot.minX, y: plot.maxY + 2))

        guard samples.count >= 2 else { return }
        let stepX = plot.width / CGFloat(history.capacity - 1)
        func point(_ i: Int, _ field: (RateSample) -> UInt64) -> NSPoint {
            NSPoint(x: plot.maxX - CGFloat(samples.count - 1 - i) * stepX,
                    y: plot.minY + CGFloat(field(samples[i])) / CGFloat(maxRate) * plot.height)
        }
        drawSeries(Self.downloadColor, samples, point, { $0.rx })
        drawSeries(Self.uploadColor, samples, point, { $0.tx })

        // 线端直接标注当前值（文本色承载文字）
        let downloadEnd = point(samples.count - 1, { $0.rx })
        let uploadEnd = point(samples.count - 1, { $0.tx })
        drawEndLabel(downloadLabel, near: downloadEnd, above: true)
        drawEndLabel(uploadLabel, near: uploadEnd, above: false)
    }

    private func drawSeries(_ color: NSColor, _ samples: [RateSample],
                            _ point: (Int, (RateSample) -> UInt64) -> NSPoint,
                            _ field: (RateSample) -> UInt64) {
        let path = NSBezierPath()
        path.move(to: point(0, field))
        for i in 1..<samples.count { path.line(to: point(i, field)) }
        color.setStroke()
        path.lineWidth = 2
        path.lineCapStyle = .round
        path.lineJoinStyle = .round
        path.stroke()
    }

    private func drawCaption(_ text: String, at point: NSPoint) {
        (text as NSString).draw(at: point, withAttributes: [
            .font: NSFont.systemFont(ofSize: 9),
            .foregroundColor: NSColor.tertiaryLabelColor,
        ])
    }

    private func drawEndLabel(_ text: String, near end: NSPoint, above: Bool) {
        let size = (text as NSString).size(withAttributes: [.font: NSFont.systemFont(ofSize: 10)])
        let x = min(end.x - size.width - 4, bounds.maxX - size.width - 2)
        let y = above ? min(end.y + 4, bounds.maxY - size.height) : max(end.y - size.height - 4, bounds.minY)
        (text as NSString).draw(at: NSPoint(x: max(bounds.minX + 2, x), y: y), withAttributes: [
            .font: NSFont.systemFont(ofSize: 10),
            .foregroundColor: NSColor.secondaryLabelColor,
        ])
    }
}
```
